cmake_minimum_required(VERSION 3.22)
project(SongCreatorVST3 VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git GIT_TAG v3.7.6_build_12)
FetchContent_MakeAvailable(vst3sdk)
smtg_enable_vst3_sdk()
set(target SongCreator)
smtg_add_vst3plugin(${target} src/factory.cpp src/plug.h src/plug.cpp src/editor.h src/editor.cpp)
smtg_target_configure_version_file(${target})
